cmake_minimum_required(VERSION 3.16)
project(delphi_rgi_core C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(dcc_rgi STATIC src/dcc_rgi.c src/dcc_identify.c src/dcc_guidance.c)
target_include_directories(dcc_rgi PUBLIC include)
add_executable(dcc_core_test tests/core_test.c)
target_link_libraries(dcc_core_test PRIVATE dcc_rgi)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dcc_rgi PRIVATE -Wall -Wextra -Werror)
  target_compile_options(dcc_core_test PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME offline_protocol_core COMMAND dcc_core_test)
add_executable(dcc_identify_test tests/identify_test.c)
target_link_libraries(dcc_identify_test PRIVATE dcc_rgi)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dcc_identify_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME offline_identify_preparation COMMAND dcc_identify_test)

add_executable(dcc_guidance_test tests/guidance_test.c)
target_link_libraries(dcc_guidance_test PRIVATE dcc_rgi)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dcc_guidance_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME offline_guidance_state COMMAND dcc_guidance_test)


# Harman-specific offline projection: no vehicle transport or firmware access.
add_library(dcc_harman STATIC ../harman/src/dcc_harman.c)
target_include_directories(dcc_harman PUBLIC ../harman/include)
target_link_libraries(dcc_harman PUBLIC dcc_rgi)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dcc_harman PRIVATE -Wall -Wextra -Werror)
endif()

add_executable(dcc_harman_test ../harman/tests/harman_test.c)
target_link_libraries(dcc_harman_test PRIVATE dcc_harman)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dcc_harman_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME offline_harman_projection COMMAND dcc_harman_test)
