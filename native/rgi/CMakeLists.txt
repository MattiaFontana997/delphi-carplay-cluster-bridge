cmake_minimum_required(VERSION 3.16)
project(delphi_rgi_core C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(dcc_rgi STATIC src/dcc_rgi.c src/dcc_identify.c)
target_include_directories(dcc_rgi PUBLIC include)
add_executable(dcc_core_test tests/core_test.c)
target_link_libraries(dcc_core_test PRIVATE dcc_rgi)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dcc_rgi PRIVATE -Wall -Wextra -Werror)
  target_compile_options(dcc_core_test PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME offline_protocol_core COMMAND dcc_core_test)
add_executable(dcc_identify_test tests/identify_test.c)
target_link_libraries(dcc_identify_test PRIVATE dcc_rgi)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dcc_identify_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME offline_identify_preparation COMMAND dcc_identify_test)
